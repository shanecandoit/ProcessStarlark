def eval(text):
    """Return the sum as an integer, calling RomanParse for each term."""
    terms = text.split(" + ")
    if len(terms) != 2:
        fail("expected two Roman numerals separated by ' + '")
    total = RomanParse.parse(terms[0]) + RomanParse.parse(terms[1])
    return total

def parse(text):
    """Parse a numeral; this small fixture does not validate Roman grammar."""
    digits = {"i": 1, "v": 5, "x": 10}
    value = 0
    previous = 0
    for index in range(len(text) - 1, -1, -1):  # read right to left
        digit = digits[text[index]]
        value += -digit if digit < previous else digit  # "iv" means 5 - 1
        previous = digit
    return value
